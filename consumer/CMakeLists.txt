# a CMake Pd external on K2FFT: add_subdirectory w/ -DK2FFT_SOURCE=<K2FFT>, else find_package (CMAKE_PREFIX_PATH)
cmake_minimum_required(VERSION 3.21)
project(fftcheck C)
if(K2FFT_SOURCE)
    add_subdirectory("${K2FFT_SOURCE}" K2FFT)
else()
    find_package(K2FFT CONFIG REQUIRED)
endif()
add_library(fftcheck MODULE fftcheck.c)
target_include_directories(fftcheck PRIVATE "${PD_INCLUDE}")
target_link_libraries(fftcheck PRIVATE K2FFT::K2FFT)
set_target_properties(fftcheck PROPERTIES PREFIX "" SUFFIX ".pd_linux")
if(APPLE)
    set_target_properties(fftcheck PROPERTIES SUFFIX ".pd_darwin")
    target_link_options(fftcheck PRIVATE "LINKER:-undefined,dynamic_lookup") # Pd's symbols, from the loading Pd
endif()
